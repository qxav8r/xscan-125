--    Opens with the scanner's live values. Read Scanner re-reads.

-- ---- Startup + Initialize ------------------------------------
-- On open: Browse tool, scanner buttons disabled.
-- Initialize runs the setup checklist into fld "Data" and enables
-- the scanner buttons only when everything passes. Help stays
-- enabled so it works even when a check fails.

on openStack
   choose browse tool
   setScannerButtons false
   put "Click Initialize to check your setup and connect to the scanner." \
         into fld "Data" of card 1 of this stack
end openStack

command setScannerButtons pOn
   local tB
   repeat for each item tB in kScannerBtns
      if there is a button tB of card 1 of this stack then
         set the enabled of button tB of card 1 of this stack to pOn
      end if
   end repeat
end setScannerButtons

command initializeScanner
   local tReport, tOK, tTool, tLib, tResult, tLine
   setScannerButtons false
   put true into tOK
   put "INITIALIZE" & cr & cr into tReport

   -- 1. Homebrew
   if there is a file "/opt/homebrew/bin/brew" or there is a file "/usr/local/bin/brew" then
      put "✓ Homebrew" & cr after tReport
   else
      put "✗ Homebrew not found -- install it from https://brew.sh" & cr after tReport
      put false into tOK
   end if

   -- 2. libusb
   put libUsbPath() into tLib
   if tLib is not empty and there is a file (tLib & "/libusb-1.0.dylib") then
      put "✓ libusb" & cr after tReport
   else
      put "✗ libusb not found -- in Terminal run:  brew install libusb" & cr after tReport
      put false into tOK
   end if

   -- 3. bc125at-tool folder (asks once if not saved yet)
   put tReport into fld "Data" of card 1 of this stack
   put toolPath() into tTool
   if tTool is empty then
      put "✗ bc125at-tool folder not set -- click Initialize and choose it" & cr after tReport
      put false into tOK
   else
      put "✓ bc125at-tool folder: " & tildePath(tTool) & cr after tReport
      -- 4. its Python environment
      if there is a file (tTool & "/.venv/bin/python3") then
         put "✓ bc125at-tool Python environment" & cr after tReport
      else
         put "✗ Python environment missing -- in Terminal:  cd " & quote & tTool & quote & \
               "  then  ./setup.sh" & cr after tReport
         put false into tOK
      end if
   end if

   if not tOK then
      put cr & "Fix the items marked ✗, then click Initialize again." after tReport
      put tReport into fld "Data" of card 1 of this stack
      exit initializeScanner
   end if

   -- 5. Scanner
   put tReport & cr & "Connecting to scanner..." into fld "Data" of card 1 of this stack
   wait 0 with messages
   put runTool("info") into tResult
   if tResult contains "Model:" then
      repeat for each line tLine in tResult
         if tLine begins with "Model:" or tLine begins with "Firmware:" then
            put "✓ " & word 1 to -1 of tLine & cr after tReport
         end if
      end repeat
      put cr & "Ready." after tReport
      setScannerButtons true
   else
      put "✗ Scanner not responding -- check it is powered on and plugged in," & cr & \
            "   then click Initialize again (the first try after plugging in" & cr & \
            "   sometimes fails)." & cr after tReport
   end if
   put tReport into fld "Data" of card 1 of this stack
end initializeScanner

   put "Reading settings from the scanner..." into fld "Data" of card 1 of stack "xscan-125"
   wait 0 with messages
   if not settingsReadScanner() then
      put "Could not read settings from the scanner. Check it's on and connected, then try again." \

-- Read the scanner's settings into sSetOrig; true if it worked
function settingsReadScanner
   local tResult, tLine, tKey
   put runTool("settings show") into tResult
   if not (tResult contains "volume:") then return false
   put empty into sSetOrig
   repeat for each line tLine in tResult
      put word 1 of tLine into tKey
      if char -1 of tKey is ":" then
         delete char -1 of tKey
         put word 2 of tLine into sSetOrig[tKey]
      end if
   end repeat
   put the time into sSetRead
   return true
end settingsReadScanner

function numberList pFrom, pTo
   local tList, i
   repeat with i = pFrom to pTo
      put i & comma after tList
   end repeat
   delete char -1 of tList
   return tList
end numberList

-- Display name of a setting ("wxalert" -> "Weather alert")
function settingName pKey
   local tRow
   set the itemDelimiter to "|"
   repeat for each line tRow in settingsRows()
      if item 1 of tRow is pKey then return item 2 of tRow
   end repeat
   return pKey
end settingName

-- Display label of a value ("wxalert", 0 -> "Off"; popups -> the number)
function settingLabel pKey, pVal
   local tRow, tVals, tNames, i
   set the itemDelimiter to "|"
   repeat for each line tRow in settingsRows()
      if item 1 of tRow is pKey then
         put item 5 of tRow into tVals
         put item 6 of tRow into tNames
         set the itemDelimiter to comma
         if tNames is empty then return pVal
         repeat with i = 1 to the number of items of tVals
            if item i of tVals is pVal then return item i of tNames
         end repeat
         return pVal
      end if
   end repeat
   return pVal
end settingLabel

-- Current choices in the palette: key -> value
function settingsChoices
   local tChoices, tBi, tB, tKey
   repeat with tBi = 1 to the number of buttons of cd 1 of stack kSetStack
      put the long id of btn tBi of cd 1 of stack kSetStack into tB
      put the uKey of tB into tKey
      if tKey is not empty then
         if the style of tB is "radioButton" then
            if the hilite of tB then
               put the uVal of tB into tChoices[tKey]
            end if
         else
            put the uVal of tB into tChoices[tKey]
         end if
      end if
   end repeat
   return tChoices
end settingsChoices

-- Number of choices that differ from the last scanner read
function settingsChangeCount
   local tChoices, tKey, tN
   put settingsChoices() into tChoices
   put 0 into tN
   repeat for each key tKey in tChoices
      if tChoices[tKey] is not sSetOrig[tKey] then add 1 to tN
   end repeat
   return tN
end settingsChangeCount

-- Set every control to the values in sSetOrig
command settingsShowValues
   local tBi, tB, tKey
   repeat with tBi = 1 to the number of buttons of cd 1 of stack kSetStack
      put the long id of btn tBi of cd 1 of stack kSetStack into tB
      put the uKey of tB into tKey
      if tKey is not empty then
         if the style of tB is "radioButton" then
            set the hilite of tB to (the uVal of tB is sSetOrig[tKey])
         else
            set the label of tB to sSetOrig[tKey]
            set the uVal of tB to sSetOrig[tKey]
         end if
      end if
   end repeat
end settingsShowValues

-- Summary line: when read, how many unsent changes
on settingsRefresh
   local tN
   if there is not a stack kSetStack then exit settingsRefresh
   put settingsChangeCount() into tN
   if tN = 0 then
      set the text of fld "summary" of cd 1 of stack kSetStack to "Read from scanner" && sSetRead & " · no changes"
   else if tN = 1 then
      set the text of fld "summary" of cd 1 of stack kSetStack to "Read from scanner" && sSetRead & " · 1 change"
   else
      set the text of fld "summary" of cd 1 of stack kSetStack to "Read from scanner" && sSetRead & " · " & tN & " changes"
   end if
end settingsRefresh

-- Read Scanner button: update every control from the scanner
on settingsReread
   local tN
   if there is not a stack kSetStack then exit settingsReread
   put settingsChangeCount() into tN
   if tN > 0 then
      answer "Discard your" && tN && "unsaved change(s) and read the scanner again?" with "Cancel" or "Read Scanner"
      if it is "Cancel" then exit settingsReread
   end if
   set the text of fld "summary" of cd 1 of stack kSetStack to "Reading the scanner..."
   wait 0 with messages
   if not settingsReadScanner() then
      set the text of fld "summary" of cd 1 of stack kSetStack to "Couldn't read the scanner -- check it's on and connected."
      exit settingsReread
   end if
   settingsShowValues
   settingsRefresh
end settingsReread

-- Apply: send only what changed (band plan first, then top to
-- bottom), re-read the scanner, report kept / not kept. The palette
-- stays open and shows the scanner's real values afterwards.
on settingsApply
   local tChoices, tOrder, tRow, tKey, tChanged, tResult, tReport
   local tKept, tLost, tSendErr, tN
   if there is not a stack kSetStack then exit settingsApply
   put settingsChoices() into tChoices

   -- order: band plan first (it can reset other settings)
   put "bandplan" into tOrder
   set the itemDelimiter to "|"
   repeat for each line tRow in settingsRows()
      if item 1 of tRow is not "bandplan" then put comma & item 1 of tRow after tOrder
   end repeat
   set the itemDelimiter to comma

   put empty into tChanged
   repeat for each item tKey in tOrder
      if tChoices[tKey] is not sSetOrig[tKey] then put tKey & comma after tChanged
   end repeat
   if tChanged is empty then
      set the text of fld "summary" of cd 1 of stack kSetStack to "No changes to send."
      exit settingsApply
   end if
   delete char -1 of tChanged

   -- send each change --
   disable btn "btnApply" of cd 1 of stack kSetStack
   put empty into tSendErr
   repeat for each item tKey in tChanged
      set the text of fld "summary" of cd 1 of stack kSetStack to "Setting" && settingName(tKey) & "..."
      wait 0 with messages
      put runTool("settings" && tKey && tChoices[tKey]) into tResult
      if not (tResult contains "Set ") then put word 1 to -1 of tResult into tSendErr[tKey]
   end repeat

   -- re-read and check what the scanner kept --
   set the text of fld "summary" of cd 1 of stack kSetStack to "Checking the scanner..."
   wait 0 with messages
   if not settingsReadScanner() then
      enable btn "btnApply" of cd 1 of stack kSetStack
      set the text of fld "summary" of cd 1 of stack kSetStack to "Sent, but couldn't read the scanner back to check."
      exit settingsApply
   end if

   put 0 into tKept
   put 0 into tLost
   put "SETTINGS  (applied" && sSetRead & ")" & cr & cr into tReport
   repeat for each item tKey in tChanged
      if sSetOrig[tKey] is tChoices[tKey] then
         add 1 to tKept
         put "✓ " & settingName(tKey) & ": " & settingLabel(tKey, tChoices[tKey]) & cr after tReport
      else
         add 1 to tLost
         put "⚠ " & settingName(tKey) & ": asked for " & settingLabel(tKey, tChoices[tKey]) & \
               ", scanner has " & settingLabel(tKey, sSetOrig[tKey]) after tReport
         if tSendErr[tKey] is not empty then put "  (" & tSendErr[tKey] & ")" after tReport
         put cr after tReport
      end if
   end repeat
   put tReport into fld "Data" of card 1 of stack "xscan-125"

   -- palette now shows the scanner's real values --

-- ---- Channel report (Read Banks) ------------------------------
-- Export -> CSV on the Desktop, then an HTML page built from that
-- CSV, opened with launch document (the browser can read the
-- Desktop; it can't read HXT's temporary folder).

command viewChannels
   local tToday, tBase, tCSV, tHTML, tResult, tInfo, tLine, tModel, tFirm
   local tBanksOn, tText, tRows, tCount
   put isoToday() into tToday
   put specialFolderPath("desktop") & "/xScan-125_Channels_" & tToday into tBase
   put tBase & ".csv" into tCSV
   put tBase & ".html" into tHTML

   -- 1. export to CSV --
   put "Reading all 500 channels from the scanner (about a minute)..." into fld "Data" of card 1 of stack "xscan-125"

   if not (tResult contains "Exported") then
      put "Export failed:" & cr & tResult into fld "Data" of card 1 of stack "xscan-125"

         "✓ CSV:  Desktop/xScan-125_Channels_" & tToday & ".csv" & cr & \

         into fld "Data" of card 1 of stack "xscan-125"

-- "2026-11-21"
function isoToday
   local tNow
   put the seconds into tNow
   convert tNow to dateItems
   return item 1 of tNow & "-" & format("%02d", item 2 of tNow) & "-" & format("%02d", item 3 of tNow)
end isoToday

-- Build the HTML page from bc125at-tool's export CSV
-- (columns: channel,name,frequency,modulation,tone,tone_code,
--  delay,lockout,priority,bank)
function channelsHTML pCSV, pModel, pFirm, pBanksLine, pWhen
   local tH, tCols, tF, i, tLine, tBank, tByBank, tCnt, tOrder, b, tOn, tFlags

      put csvSplit(tRow) into tLine

   put "h1 { font-size: 22px; margin: 0 0 4px; }" & return after tH

   put "input { font: inherit; padding: 4px 8px; width: 280px; margin-bottom: 6px; }" & return after tH
   put "h2 { font-size: 16px; margin: 26px 0 6px; }" & return after tH
   put "h2 span { color: GrayText; font-weight: normal; }" & return after tH
   put "table { border-collapse: collapse; width: 100%; max-width: 920px; }" & return after tH
   put "th { text-align: left; font-weight: 600; padding: 4px 8px; border-bottom: 1px solid GrayText; }" & return after tH
   put "td { padding: 3px 8px; white-space: nowrap; }" & return after tH
   put "td.n { text-align: right; font-variant-numeric: tabular-nums; }" & return after tH
   put "tbody tr:nth-child(even) { background: rgba(127,127,127,0.12); }" & return after tH
   put "@media print { input { display: none; } section { break-inside: avoid; } }" & return after tH
   put "</style>" & return after tH
   put "<script>function f(q){q=q.toLowerCase();" & \
         "document.querySelectorAll('tbody tr').forEach(function(r){" & \
         "r.style.display=r.textContent.toLowerCase().indexOf(q)<0?'none':'';});" & \
         "document.querySelectorAll('section').forEach(function(s){" & \
         "var v=s.querySelectorAll('tbody tr:not([style*=none])').length;" & \
         "s.style.display=v?'':'none';});}</script>" & return after tH
   put "</head><body>" & return after tH

   put "<h1>xScan-125 Channels</h1>" & return after tH
   put "<p class='meta'>" & htmlEsc(pModel) && "· Firmware" && htmlEsc(pFirm) && "· " & \
         tTotal && "programmed channels · read" && htmlEsc(pWhen) & "</p>" & return after tH
   put "<input type='search' placeholder='Filter by name, frequency, tone...' oninput='f(this.value)'>" & return after tH

   -- banks in scanner order: 1-9 then 0 (channels 451-500) --
   put "1,2,3,4,5,6,7,8,9,0" into tOrder
   set the itemDelimiter to comma
   repeat for each item b in tOrder
      if tCnt[b] is empty then next repeat
      if pBanksLine contains ("✓ B" & b) then
         put "scanning on" into tOn
      else
         put "scanning off" into tOn
      end if
      put "<section><h2>Bank " & b after tH
      if b is 0 then put " <span>(channels 451–500)</span>" after tH
      put " <span>· " & tCnt[b] && "channels ·" && tOn & "</span></h2>" & return after tH
      put "<table><thead><tr><th>CH</th><th>Name</th><th>Frequency</th><th>Mod</th>" & \
            "<th>Tone</th><th>Delay</th><th>Flags</th></tr></thead><tbody>" & return after tH
      set the itemDelimiter to tab
      repeat for each line tLine in tByBank[b]
         if tLine is empty then next repeat
         put item tF["tone"] of tLine into tTone
         if tTone is "None" then put empty into tTone
         put item tF["frequency"] of tLine into tFreq
         if tFreq is a number then put format("%.4f", tFreq) into tFreq
         put empty into tFlags
         if item tF["priority"] of tLine is "True" then put "PRI " after tFlags
         if item tF["lockout"] of tLine is "True" then put "L/O" after tFlags
         put "<tr><td class='n'>" & item tF["channel"] of tLine & "</td>" & \
               "<td>" & htmlEsc(item tF["name"] of tLine) & "</td>" & \

   return tH

-- One CSV line -> tab-delimited fields (handles "quoted, fields")
function csvSplit pLine
   local tFields, tField, tInQ, i, n, c
   put false into tInQ
   put the number of chars of pLine into n
   put 1 into i
   repeat while i <= n
      put char i of pLine into c
      if tInQ then
         if c is quote then
            if char (i + 1) of pLine is quote then
               put quote after tField
               add 1 to i
            else
               put false into tInQ
            end if
         else
            put c after tField
         end if
      else
         if c is quote then
            put true into tInQ
         else if c is comma then
            put tField & tab after tFields
            put empty into tField
         else
            put c after tField
         end if
      end if
      add 1 to i
   end repeat
   put tField after tFields
   return tFields
end csvSplit

function htmlEsc pText
   replace "&" with "&amp;" in pText
   replace "<" with "&lt;" in pText
   replace ">" with "&gt;" in pText
   replace quote with "&quot;" in pText
   return pText
end htmlEsc

-- ---- bc125at-tool location -----------------------------------
-- Found fresh each session (cached in gToolPath until quit):
--   1) gToolPath, if already found this session
--   2) usual places, then a Spotlight search
--   3) only if both fail, ask the user (with a hint)
-- A folder only counts if it has the bc125at package AND its
-- Python environment (.venv/bin/python3), i.e. setup.sh was run.
-- Nothing is saved -- see HELP NOTES in the header for the
-- trade-off.

function toolPath
   if isToolFolder(gToolPath) then return gToolPath   -- already found this session

   put findToolFolder() into gToolPath
   if gToolPath is not empty then return gToolPath

   answer folder "xScan-125 couldn't find bc125at-tool on this Mac." & cr & \
         "Select the bc125at-tool folder -- the one with setup.sh and a" & cr & \
         "bc125at folder inside. It's usually in your home folder."
   if it is empty then
      put empty into gToolPath
      return empty
   end if
   if not isToolFolder(it) then
      answer "That folder isn't a ready-to-use bc125at-tool." & cr & cr & \
            "If you just downloaded it, open Terminal, cd into that folder" & cr & \
            "and run ./setup.sh, then click Initialize again." & cr & cr & \
            "Download: https://github.com/costamesatechsolutions/bc125at-tool"
      put empty into gToolPath
      return empty
   end if
   put it into gToolPath
   return gToolPath
end toolPath

-- True if pPath is a bc125at-tool folder that has been set up
function isToolFolder pPath
   if pPath is empty then return false
   if there is not a folder (pPath & "/bc125at") then return false
   return there is a file (pPath & "/.venv/bin/python3")
end isToolFolder

-- Usual places first, then Spotlight (any folder named bc125at-tool*,
-- which also catches "bc125at-tool-main" from a GitHub zip download)
function findToolFolder
   local tHome, tList, tPath, tFound
   put specialFolderPath("home") into tHome
   put tHome & "/bc125at-tool" & cr & \
         tHome & "/Documents/bc125at-tool" & cr & \
         tHome & "/Desktop/bc125at-tool" & cr & \
         tHome & "/Downloads/bc125at-tool" & cr & \
         tHome & "/Downloads/bc125at-tool-main" & cr & \
         tHome & "/Documents/GitHub/bc125at-tool" & cr & \
         tHome & "/Documents/GitHub/xscan-125/bc125at-tool" & cr & \
         stackFolder() & "/bc125at-tool" into tList
   repeat for each line tPath in tList
      if isToolFolder(tPath) then return tPath
   end repeat

   put shell("mdfind 'kMDItemFSName == " & quote & "bc125at-tool*" & quote & "c' 2>/dev/null") into tFound
   repeat for each line tPath in tFound
      if isToolFolder(tPath) then return tPath
   end repeat
   return empty
end findToolFolder

-- Folder this stack file is in
function stackFolder
   local tPath
   put the effective filename of this stack into tPath
   set the itemDelimiter to "/"
   return item 1 to -2 of tPath
end stackFolder

-- Homebrew libusb: Apple Silicon first, then Intel. No $(brew ...)
-- because shell() doesn't load your login profile.
function libUsbPath
   if there is a folder "/opt/homebrew/opt/libusb/lib" then return "/opt/homebrew/opt/libusb/lib"
   if there is a folder "/usr/local/opt/libusb/lib" then return "/usr/local/opt/libusb/lib"
   return empty
end libUsbPath

-- ---- One shared call to bc125at-tool -------------------------
-- pArgs = everything after "bc125at", e.g. "info",
-- "channels --bank 3", "import <quoted path>".
-- Returns UTF-8-decoded output with progress-bar lines ("Writing [",
-- "Reading [") removed;
-- empty if no tool folder or no output.
-- Shows the watch cursor, so every scanner button gets it for free.

function runTool pArgs
   local tTool, tLib, tCmd, tResult, tClean, tLine
   set the cursor to watch   -- busy cursor while the scanner works (resets when handlers finish)
   put toolPath() into tTool
   if tTool is empty then return empty

   put "cd " & quote & tTool & quote & " && " into tCmd
   put libUsbPath() into tLib
   if tLib is not empty then
      put "DYLD_LIBRARY_PATH=" & quote & tLib & quote & " " after tCmd
   end if
   put "./.venv/bin/python3 -m bc125at " & pArgs after tCmd

   put textDecode(shell(tCmd), "UTF8") into tResult
   repeat for each line tLine in tResult
      if not (tLine contains "Writing [" or tLine contains "Reading [") then
         put tLine & cr after tClean
      end if
   end repeat
   return tClean
end runTool

function inventoryOneStack pStack
   local tOut, c, x, tCard, tName

   put "[ STACK ] " & pStack & return into tOut
   put scriptBlock("STACK: " & pStack, the script of stack pStack) & return after tOut
   if the behavior of stack pStack is not empty then
      put behaviorBlock("STACK: " & pStack, the behavior of stack pStack) after tOut
   end if

   put "[ BACKGROUNDS ] " & the number of bgs of stack pStack & return after tOut
   repeat with x = 1 to the number of bgs of stack pStack
      put " bg " & the short name of bg x of stack pStack & return after tOut
   end repeat
   put return after tOut

   put "[ CARDS ] " & the number of cards of stack pStack & " total" & return & return after tOut
   repeat with c = 1 to the number of cards of stack pStack
      put the short name of card c of stack pStack into tCard
      put "[ CARD " & c & " ] " & tCard & return & return after tOut

      put "-- card script --" & return after tOut
      put scriptBlock("CARD " & c & ": " & tCard, the script of card c of stack pStack) & return after tOut

      -- groups
      put "-- groups --" & return after tOut
      repeat with x = 1 to the number of groups of card c of stack pStack
         put " group " & the short name of group x of card c of stack pStack after tOut
         if the label of group x of card c of stack pStack is not empty then
            put " [" & the label of group x of card c of stack pStack & "]" after tOut
         end if
         if the backgroundBehavior of group x of card c of stack pStack then put " (background)" after tOut
         if seenBefore(the long id of group x of card c of stack pStack, c) then
            put sSeen["note"] & return after tOut
         else
            put return & scriptBlock("GROUP: " & the short name of group x of card c of stack pStack, \
                  the script of group x of card c of stack pStack) after tOut
            if the behavior of group x of card c of stack pStack is not empty then
               put behaviorBlock("GROUP: " & the short name of group x of card c of stack pStack, \

      end repeat

         end if
         put " loc=" & the loc of button x of card c of stack pStack after tOut
         if the short name of the owner of button x of card c of stack pStack is not tCard then
            put " (in group: " & the short name of the owner of button x of card c of stack pStack & ")" after tOut
         end if
         if seenBefore(the long id of button x of card c of stack pStack, c) then
            put sSeen["note"] & return after tOut
         else
            put return & scriptBlock("BUTTON: " & tName, the script of button x of card c of stack pStack) after tOut
            if the behavior of button x of card c of stack pStack is not empty then
               put behaviorBlock("BUTTON: " & tName, the behavior of button x of card c of stack pStack) after tOut
            end if
         end if
      end repeat
      put return after tOut

      -- fields
      put "-- fields --" & return after tOut
      repeat with x = 1 to the number of fields of card c of stack pStack
         put the short name of field x of card c of stack pStack into tName
         put " field " & tName & " loc=" & the loc of field x of card c of stack pStack after tOut
         if the short name of the owner of field x of card c of stack pStack is not tCard then
            put " (in group: " & the short name of the owner of field x of card c of stack pStack & ")" after tOut
         end if
         if seenBefore(the long id of field x of card c of stack pStack, c) then
            put sSeen["note"] & return after tOut
         else
            put return & scriptBlock("FIELD: " & tName, the script of field x of card c of stack pStack) after tOut
            if the behavior of field x of card c of stack pStack is not empty then
               put behaviorBlock("FIELD: " & tName, the behavior of field x of card c of stack pStack) after tOut
            end if
         end if
      end repeat
      put return after tOut

      -- everything else (name + script only)
      put "-- other controls --" & return after tOut
      repeat with x = 1 to the number of graphics of card c of stack pStack
         put " graphic " & the short name of graphic x of card c of stack pStack & return & \
               scriptBlock("GRAPHIC: " & the short name of graphic x of card c of stack pStack, \
               the script of graphic x of card c of stack pStack) after tOut
      end repeat
      repeat with x = 1 to the number of images of card c of stack pStack
         put " image " & the short name of image x of card c of stack pStack & return & \
               scriptBlock("IMAGE: " & the short name of image x of card c of stack pStack, \
               the script of image x of card c of stack pStack) after tOut

      repeat with x = 1 to the number of players of card c of stack pStack
         put " player " & the short name of player x of card c of stack pStack & return & \
               scriptBlock("PLAYER: " & the short name of player x of card c of stack pStack, \
               the script of player x of card c of stack pStack) after tOut
      end repeat
      repeat with x = 1 to the number of scrollbars of card c of stack pStack
         put " scrollbar " & the short name of scrollbar x of card c of stack pStack & return & \
               scriptBlock("SCROLLBAR: " & the short name of scrollbar x of card c of stack pStack, \
               the script of scrollbar x of card c of stack pStack) after tOut
      end repeat
      repeat with x = 1 to the number of widgets of card c of stack pStack
         put " widget " & the short name of widget x of card c of stack pStack & return & \
               scriptBlock("WIDGET: " & the short name of widget x of card c of stack pStack, \
               the script of widget x of card c of stack pStack) after tOut
      end repeat
      put return after tOut
   end repeat
   return tOut
end inventoryOneStack

-- Script text block + adds the script's handlers to the index
function scriptBlock pLabel, pScript
   if pScript is empty then return " (no script)" & return
   addToIndex pLabel, pScript
   return " *** HAS SCRIPT ***" & return & kRule & return & pScript & return & kRule & return
end scriptBlock

-- Behavior reference + the behavior object's own script
function behaviorBlock pLabel, pBehavior
   local tOut
   put "   behavior: " & pBehavior & return into tOut
   put scriptBlock("BEHAVIOR for " & pLabel, the script of pBehavior) after tOut
   return tOut
end behaviorBlock

-- True if this object was already listed (shared group on a later card)
function seenBefore pLongID, pCardNum
   if sSeen[pLongID] is not empty then
      put " (listed on card " & sSeen[pLongID] & ")" into sSeen["note"]
      return true
   end if
   put pCardNum into sSeen[pLongID]
   return false
end seenBefore

command addToIndex pLabel, pScript
   local tLine, tW1
   put return & "── " & pLabel & return after sIndex
   repeat for each line tLine in pScript
      put word 1 of tLine into tW1
      if tW1 is "private" then put word 2 of tLine into tW1
      if tW1 is among the items of "on,function,command,getProp,setProp,before,after" then
         if word 1 of tLine is "private" then
            put "  " & word 1 to 3 of tLine & return after sIndex
         else
            put "  " & word 1 to 2 of tLine & return after sIndex
         end if
      end if
   end repeat
end addToIndex

-- Swap the home folder for ~ (keeps usernames out of the repo)
function tildePath pPath
   local tHome
   put specialFolderPath("home") into tHome
   if tHome is not empty and pPath begins with tHome then
      put "~" & char (length(tHome) + 1) to -1 of pPath into pPath
   end if
   return pPath
end tildePath
